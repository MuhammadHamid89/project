// Action button
// App.jsx file:

import { useDispatch,useSelector } from "react-redux"
import { increment,decrement,reset,addFive,addAmount } from "./features/counter/counterSlice"
export default function App(){
  const dispatch=useDispatch()  //user action perfom through dispatch function
  const count=useSelector((state)=>state.counter.count)
  return(
    <>
    <h1>redux set up complete</h1>
    <h2>{count}</h2>
    <button onClick={()=>dispatch(increment())}>increment</button>
    <button onClick={()=>dispatch(decrement())}>decrement</button>
    <button onClick={()=>dispatch(reset())}>reset</button>
    <button onClick={()=>dispatch(addFive())}>Add 5</button>
    <button onClick={()=>dispatch(addAmount(10))}>Add 10</button>
    </>
  )
}

// store.js

import { configureStore } from "@reduxjs/toolkit";
import counterReducer from '../features/counter/counterSlice'
export const store= configureStore({
    reducer:{
        counter:counterReducer,
    },
})



// counterSlics.js
// action make and use immer not use spread operator

import { createSlice } from "@reduxjs/toolkit";
const initialState={
    count:0,
}

const counterSlice=createSlice({   //hum yha object ki surat my value d rhy hain
    name:'counter',
    initialState,
    reducers:{
        increment(state){
            state.count++;
        },
        decrement(state){
            state.count--;
        },
        reset(state){
            state.count=0;
        },
        addFive(state){
            state.count+=5
        },
        addAmount(state,action){
            state.count+=action.payload  //api call or use store update use payload
        }
    },
})
export const {increment,decrement,reset,addFive,addAmount}=counterSlice.actions  // action create who user perform
export default counterSlice.reducer
